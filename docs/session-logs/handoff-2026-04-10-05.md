### llm/client.py — Token proxy

```python
from utils.token_proxy import get_token_proxy

async def call(self, model, system, messages, ...):
    proxy = get_token_proxy()
    cache_salt = f"{model}::{agent_name}"
    # Compress and check cache on user message
    user_prompt = messages[-1]["content"]
    compressed, cached = proxy.optimize(user_prompt, model=model, salt=cache_salt)
    if cached is not None:
        return cached
    messages[-1]["content"] = compressed
    result = await self._call_with_retry(...)
    if result.success:
        proxy.set_cached(compressed, result, model=model, salt=cache_salt)
    return result
```
